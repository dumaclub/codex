MySQL 8.0 이상이라면 `ROW_NUMBER()`를 이용해 `mailId`를 10000부터 순차 생성할 수 있습니다.

```sql
INSERT INTO ast_game21.mail
(
    mailId,
    uId,
    charId,
    mailType,
    fromCharId,
    fromCharName,
    fromTableId,
    fromGuildName,
    fromCrestId,
    state,
    title,
    message,
    attachedItemId,
    attachedItemCount,
    regDate,
    reservedDate,
    expireDate,
    versionDate,
    isDelete,
    accountType
)
SELECT
    9999 + ROW_NUMBER() OVER (ORDER BY bu.uId, bu.idx) AS mailId,
    bu.uId                                             AS uId,
    0                                                  AS charId,
    1                                                  AS mailType,
    0                                                  AS fromCharId,
    'Astellia'                                         AS fromCharName,
    0                                                  AS fromTableId,
    ''                                                 AS fromGuildName,
    0                                                  AS fromCrestId,
    0                                                  AS state,
    'UI_Privilege Daily Reward_Title'                  AS title,
    'UI_Privilege Daily Reward_Desc'                   AS message,
```
