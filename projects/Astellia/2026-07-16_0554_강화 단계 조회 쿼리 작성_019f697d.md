UPDATE ast_game01.collection AS c
JOIN ast_info.CollectionData AS cd
    ON cd.Id = c.collectionId
JOIN ast_info.CollectionItemData AS cid
    ON cid.Id = cd.CollectionItemDataId
JOIN (
    SELECT
        idx,
        TRIM(BOTH ',' FROM IFNULL(collections, '')) AS oldCollections
    FROM ast_game01.collection
) AS src
    ON src.idx = c.idx
SET
    c.collectionCount = CONCAT(
        IF(cid.CollectionItemId1 > 0, CONCAT(IF(FIND_IN_SET(CAST(cid.CollectionItemId1 AS CHAR), src.oldCollections) > 0, '1', '0'), ','), ''),
