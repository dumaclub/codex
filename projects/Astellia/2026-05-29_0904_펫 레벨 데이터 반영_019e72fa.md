```csharp
public int GetPetLevel(int petLevelGroupId, int petLevel, long petExp, int awakenMaxLevel)
{
    var chgPetLevel = petLevel;

    var petLevelGroupList = DataManager.Get<LoaderPetData>().GetPetLevelGroupData(petLevelGroupId);
    if (petLevelGroupList == null || petLevelGroupList.Count < 1)
        return petLevel;

    foreach (var petLevelData in petLevelGroupList.OrderBy(x => x.Level))
    {
        if (petLevelData.Level < petLevel)
            continue;

        // 현재 각성단계의 최대레벨까지만 확인한다.
        if (petLevelData.Level > awakenMaxLevel)
            break;

        if (petExp < petLevelData.AccumulateExp)
            break;

        chgPetLevel = petLevelData.Level;
    }

    return chgPetLevel > petLevel ? chgPetLevel : petLevel;
}
```
