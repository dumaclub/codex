CREATE TEMPORARY TABLE tmp_monster_check_6221 AS
SELECT
    x.backupIdx,
    x.uId,
    x.collectionId,
    x.itemPosition,
    x.itemId,
    x.backupCount,
    x.currentIdx,
    x.currentState,
    x.currentCollections,
    x.currentCollectionCount,
    x.currentItemPosition,
    x.currentCountValueCount,
    CASE
        WHEN x.currentItemPosition > 0
         AND x.currentItemPosition <= x.currentCountValueCount
        THEN COALESCE(
            CAST(
                NULLIF(
                    SUBSTRING_INDEX(
                        SUBSTRING_INDEX(
                            TRIM(BOTH ',' FROM x.currentCollectionCount),
                            ',',
                            x.currentItemPosition
                        ),
                        ',',
                        -1
                    ),
                    ''
                ) AS UNSIGNED
            ),
            0
        )

    END AS currentCount

        p.*,
        c.idx AS currentIdx,
        c.state AS currentState,
        c.collections AS currentCollections,
        c.collectionCount AS currentCollectionCount,
        FIND_IN_SET(

    FROM

            backup.uId,
            backup.collectionId,
